// 用户点击开始自由聊天
Client.startScene(
        StartSceneRequest
)
SceneSession session =
        SceneSessionFactory.createSceneSession(
                SceneType.FREE_CHAT
        )
// 设置AI角色
session.setAgentProfile(
        AgentProfile.builder()
            .setAgentType(
                AgentType.ENGLISH_COACH
            )
            .setTeachingStyle(
                "friendly"
            )
            .build()
)

// 设置用户本次会话上下文
session.setSceneContext(
        SceneContext.builder()
            .setSceneType(
                SceneType.FREE_CHAT
            )
            .setDifficultyLevel(
                "B1"
            )
            .setUserPreference(
                userPreference
            )
            .build()
)

// 获取Prompt模板
// 例如：
//
// FreeChatPromptTemplate
//
// 你是一名英语口语陪练老师。
//
// 用户当前英语水平：{level}
//
// 教学风格：{style}
//
// 规则：
// 1. 使用自然英语交流。
// 2. 回复控制2-3句话。
// 3. 用户中英文混合时理解真实含义。
// 4. 根据用户水平调整表达。
// 5. 不主动结束聊天。
PromptTemplate template =
        PromptTemplateRepository.getTemplate(
                SceneType.FREE_CHAT
        )
SessionPrompt prompt =
        PromptBuilder.buildPrompt(
                PromptBuildContext.builder()
                    .setTemplate(
                            template
                    )
                    .setSceneContext(
                            session.getSceneContext()
                    )
                    .setAgentProfile(
                            agentProfile
                    )
                    .build()
        )

// 设置Prompt
session.setPrompt(
        prompt
)

// 获取Realtime模型适配器
RealtimeAdapter adapter =
        RealtimeAdapterRegistry.getAdapter(
                RealtimeProviderType.QWEN
        )

// 设置模型Adapter
session.setRealtimeAdapter(
        adapter
)

// 创建Realtime模型会话
RealtimeSession realtimeSession = session.createRealtimeSession()


// 调用模型厂商
RealtimeAdapter.createSession(
        RealtimeSessionConfig.builder()
            .setUserId(
                    userId
            )
            .setSystemPrompt(
                    prompt
            )
            .setVoiceId(
                    voicePreference
            )
            .build()
)

// 返回：
RealtimeSession {

    token;
    endpoint;
}
