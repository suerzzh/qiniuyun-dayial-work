# AI 场景会话服务体系——雅思模拟模块伪代码

> 本文以《自由聊天伪代码》的分层树状结构、公共抽象类和 Java 风格伪代码为标准，将自由聊天场景替换为 IELTS Speaking 模拟考试场景。
>
> 复用原则：`SessionService`、`AuthService`、`ProfileService`、`SessionIdGenerator`、`RealtimeConnectionService`、`SessionStateStore`、`SessionBindingTimeoutService`、`UsageQuotaService`、`ScenePromptService` 和 `AbstractSceneSession` 保持为公共能力；新增 IELTS 题库、组卷、状态机、Attempt、Turn、PCM、评分和报告能力。

@Service
public class IeltsMockSessionService extends SessionService {

    // ========================================================
    // 1. 启动 IELTS Speaking 模拟考试
    // 前端通过 HTTP 发送 Offer SDP 和考试配置
    // ========================================================

    public StartIeltsMockResult start(
            StartIeltsMockCommand command) {

        // 获取当前用户和用户画像
        String userId =
            AuthService.getCurrentUserId();

        UserProfile profile =
            ProfileService.getProfile(userId);


        // 获取 IELTS 模拟场景配置并校验访问权限
        SceneConfig sceneConfig =
            SceneService.getByType(
                SceneType.IELTS_MOCK
            );

        SceneService.validateAccess(
            userId,
            sceneConfig
        );

        UsageQuotaService.validate(
            userId,
            SceneType.IELTS_MOCK
        );


        // 加载并校验 IELTS 题库
        IeltsQuestionBank questionBank =
            IeltsQuestionBankService.loadQuestionBank(
                command.getBankVersion()
            );


        // 根据考试模式生成不可变 Paper Snapshot
        PaperSnapshot paperSnapshot =
            IeltsPaperAssemblerService.assemble(
                AssemblePaperCommand.builder()
                    .setQuestionBank(questionBank)
                    .setMode(command.getMode())
                    .setSelectedPart(command.getSelectedPart())
                    .setTimingProfile(
                        command.getTimingProfile()
                    )
                    .setRecentQuestionIds(
                        IeltsLearningRecordService
                            .getRecentQuestionIds(userId)
                    )
                    .build()
            );


        // 创建本地业务会话
        String localSessionId =
            SessionIdGenerator.generate();

        IeltsMockSceneSession session =
            new IeltsMockSceneSession(
                localSessionId,
                userId
            );


        // 创建整场考试 Attempt
        IeltsAttempt attempt =
            IeltsAttemptService.createAttempt(
                CreateAttemptCommand.builder()
                    .setUserId(userId)
                    .setLocalSessionId(localSessionId)
                    .setMode(command.getMode())
                    .setPaperSnapshot(paperSnapshot)
                    .build()
            );


        // 创建确定性考试状态机
        IeltsExamState examState =
            IeltsExamFlowService.createState(
                paperSnapshot,
                SessionPolicy.builder()
                    .setCaptionsEnabled(
                        command.isCaptionsEnabled()
                    )
                    .setRecordingEnabled(
                        command.isRecordingEnabled()
                    )
                    .setStrictMock(
                        command.getMode()
                            == IeltsMode.FULL_MOCK
                    )
                    .build()
            );


        // 生成 IELTS Realtime 考官 Prompt
        SessionPrompt examinerPrompt =
            IeltsExaminerPromptService.build(
                profile,
                sceneConfig,
                paperSnapshot
            );


        // 设置会话场景和运行参数
        session.setSceneId(
            sceneConfig.getSceneId()
        );

        session.setProviderType(
            sceneConfig.getProviderType()
        );

        session.setPrompt(examinerPrompt);

        session.setVoiceId(
            profile.getVoiceId() != null
                ? profile.getVoiceId()
                : sceneConfig.getDefaultVoiceId()
        );

        session.setAttemptId(
            attempt.getAttemptId()
        );

        session.setExamMode(
            command.getMode()
        );

        session.setTimingProfile(
            paperSnapshot.getTimingProfile()
        );

        session.setPaperSnapshot(
            paperSnapshot
        );

        session.setExamState(
            examState
        );


        // 预占用户额度
        UsageQuotaService.reserve(
            userId,
            localSessionId,
            SceneType.IELTS_MOCK
        );


        // 保存会话和 Attempt
        SessionStateStore.save(session);
        IeltsAttemptService.save(attempt);


        try {

                    session.getProviderType(),
                    command.getOfferSdp(),
                    examinerPrompt
                );

            session.markConnecting();
            SessionStateStore.save(session);

                localSessionId,
                BIND_TIMEOUT
            );

            return StartIeltsMockResult.builder()
                .setLocalSessionId(localSessionId)
                .setAttemptId(attempt.getAttemptId())
                .setAnswerSdp(answerSdp)

                    + "?attempt_id="
                    + attempt.getAttemptId()

                .setSystemPrompt(examinerPrompt)
                .setVoiceId(session.getVoiceId())

                exception.getMessage()
            );

            UsageQuotaService.release(
                localSessionId
            );

            throw exception;
        }

    // ========================================================
    // 2. 统一处理 IELTS 会话和考试事件
    // ========================================================

    public void handleEvent(
            String localSessionId,
            SessionEvent event) {

        IeltsMockSceneSession session =
            SessionStateStore.get(localSessionId);

        switch (event.getType()) {

            // Qwen Realtime 创建模型会话
            case PROVIDER_SESSION_CREATED -> {

                session.bindProviderSession(
                    event.getProviderSessionId()
                );

                SessionBindingTimeoutService.cancel(
                    localSessionId
                );

                UsageQuotaService.startMetering(
                    localSessionId
                );

                SessionStateStore.save(session);
            }


            // Qwen Realtime 完成 Prompt、音色和 VAD 配置
            case PROVIDER_SESSION_UPDATED -> {

                session.activate();

                session.setExamState(
                    IeltsExamFlowService.transition(
                        session.getExamState(),
                        IeltsExamEvent.START
                    )
                );

                SessionStateStore.save(session);

                // 前端收到状态后发送固定 Opening 指令
                SessionWebSocketHandler.sendExamCommand(
                    localSessionId,
                    ExaminerCommand.playOpening()
                );
            }


            // 开始 Introduction，保存但不参与评分
            case INTRODUCTION_STARTED -> {

                IeltsTurn turn =
                    IeltsTurnService.openTurn(
                        OpenTurnCommand.builder()
                            .setAttemptId(
                                session.getAttemptId()
                            )
                            .setPart(0)
                            .setQuestionId("introduction")
                            .setQuestionText(
                                "Please introduce yourself."
                            )
                            .setTurnType(
                                TurnType.INTRODUCTION
                            )
                            .setScoringEligible(false)
                            .build()
                    );

                session.setActiveTurnId(
                    turn.getTurnId()
                );

                SessionStateStore.save(session);
            }


            // 前端通知当前 Part 已开始
            case PART_STARTED -> {

                IeltsExamFlowService.validatePart(
                    session.getExamState(),
                    event.getPart()
                );

                session.setCurrentPart(
                    event.getPart()
                );

                IeltsAttemptService.updateCurrentPart(
                    session.getAttemptId(),
                    event.getPart()
                );

                SessionStateStore.save(session);
            }


            // 前端通知状态机已经提出冻结题目
            case QUESTION_ASKED -> {

                IeltsExamFlowService
                    .validateQuestionAgainstPaper(
                        session.getPaperSnapshot(),
                        event.getPart(),
                        event.getQuestionId(),
                        event.getQuestionText()
                    );

                session.setCurrentQuestionId(
                    event.getQuestionId()
                );

                IeltsAttemptService.updateCurrentQuestion(
                    session.getAttemptId(),
                    event.getQuestionId(),
                    event.getQuestionText()
                );

                SessionStateStore.save(session);
            }


            // 打开一道题对应的回答 Turn
            case TURN_OPENED -> {

                boolean part2LongTurn =
                    event.getTurnType()
                        == TurnType.PART2_LONG_TURN;

                IeltsTurn turn =
                    IeltsTurnService.openTurn(
                        OpenTurnCommand.builder()
                            .setAttemptId(
                                session.getAttemptId()
                            )
                            .setPart(event.getPart())
                            .setQuestionId(
                                event.getQuestionId()
                            )
                            .setQuestionText(
                                event.getQuestionText()
                            )
                            .setTurnType(
                                event.getTurnType()
                            )
                            .setScoringEligible(
                                event.getPart() >= 1
                                && event.getPart() <= 3
                            )
                            .build()
                    );

                IeltsTurnAssemblerService.openTurn(
                    turn.getTurnId(),
                    event.getPart(),
                    event.getQuestionText(),
                    event.getTurnType()
                );

                session.setActiveTurnId(
                    turn.getTurnId()
                );

                SessionStateStore.save(session);
            }


            // Qwen VAD 检测到考生开始说话
            case USER_SPEECH_STARTED -> {

                String turnId =
                    session.getActiveTurnId();

                IeltsTurnService.speechStarted(
                    turnId,
                    event.getTimestampMs()
                );

                IeltsTurnAssemblerService.speechStarted(
                    turnId,
                    event.getTimestampMs()
                );
            }


            // Qwen VAD 检测到考生停止说话
            case USER_SPEECH_STOPPED -> {

                String turnId =
                    session.getActiveTurnId();

                IeltsTurnService.speechStopped(
                    turnId,
                    event.getTimestampMs()
                );

                IeltsTurnAssemblerService.speechStopped(
                    turnId,
                    event.getTimestampMs()
                );

                // 注意：Part 2 只结束 speech chunk，
                // 不结束整个 PART2_LONG_TURN。
            }


            // 保存 Qwen Realtime 原始完整转写
            case USER_TRANSCRIPT_COMPLETED -> {

                String turnId =
                    session.getActiveTurnId();

                IeltsTurnService.appendRawTranscript(
                    AppendTranscriptCommand.builder()
                        .setTurnId(turnId)
                        .setRawTranscript(
                            event.getTranscript()
                        )
                        .build()
                );
            }


            // 用户明确结束普通回答
            case TURN_COMPLETED -> {

                completeCurrentTurn(
                    session,
                    event.getCompletionReason()
                );

                advanceExamState(
                    session,
                    IeltsExamEvent.SUBMIT_ANSWER
                );
            }


            // 用户明确结束 Part 2 长回答
            case PART2_SPEAKING_COMPLETED -> {

                completeCurrentTurn(
                    session,
                    event.getCompletionReason()
                );

                advanceExamState(
                    session,
                    IeltsExamEvent.SUBMIT_ANSWER
                );
            }


            // Part 2 准备时间结束
            case PART2_PREPARATION_EXPIRED -> {

                advanceExamState(
                    session,
                    IeltsExamEvent.PREP_EXPIRED
                );
            }


            // Part 3 达到软时间限制
            case PART3_SOFT_LIMIT_REACHED -> {

                advanceExamState(
                    session,
                    IeltsExamEvent.PART3_SOFT_LIMIT
                );
            }


            // Part 3 达到硬时间限制
            case PART3_HARD_LIMIT_REACHED -> {

                if (session.getActiveTurnId() != null) {
                    completeCurrentTurn(
                        session,
                        CompletionReason.TIME_LIMIT
                    );
                }

                advanceExamState(
                    session,
                    IeltsExamEvent.PART3_HARD_LIMIT
                );

                handleSessionCompleted(
                    localSessionId
                );
            }


            // 用户正常完成整场考试
            case EXAM_COMPLETED,
                 SESSION_COMPLETE -> {

                session.complete();
                SessionStateStore.save(session);

                handleSessionCompleted(
                    localSessionId
                );
            }


            // 用户主动退出
            case EXAM_ABANDON -> {

                session.abandon();
                SessionStateStore.save(session);

                IeltsAttemptService.abandonAttempt(
                    session.getAttemptId()
                );

                IeltsTurnAssemblerService.finishStream(
                    session.getAttemptId()
                );

                UsageQuotaService.settle(
                    localSessionId
                );
            }


            // 模型、WebRTC、评分 WebSocket 或状态机异常
            case SESSION_ERROR -> {

                session.fail(
                    event.getErrorCode(),
                    event.getErrorMessage()
                );

                SessionStateStore.save(session);

                IeltsAttemptService.markTechnicalInterrupted(
                    session.getAttemptId(),
                    event.getErrorMessage()
                );

                IeltsTurnAssemblerService.finishStream(
                    session.getAttemptId()
                );

                UsageQuotaService.settle(
                    localSessionId
                );
            }


            // 规定时间内没有绑定 providerSessionId
            case BIND_TIMEOUT -> {

                if (session.getProviderSessionId() == null) {

                    session.fail(
                        "PROVIDER_BIND_TIMEOUT",
                        "Qwen Realtime 会话绑定超时"
                    );

                    SessionStateStore.save(session);

                    IeltsAttemptService
                        .markTechnicalInterrupted(
                            session.getAttemptId(),
                            "Qwen Realtime 会话绑定超时"
                        );

                    UsageQuotaService
                        .settleReservedQuota(
                            localSessionId
                        );

                    SessionWebSocketHandler
                        .sendCloseWebRtc(
                            localSessionId
                        );
                }
            }
        }
    }


    private void completeCurrentTurn(
            IeltsMockSceneSession session,
            CompletionReason reason) {

        String turnId =
            session.getActiveTurnId();

        if (turnId == null) {
            return;
        }

        long completedAt =
            System.currentTimeMillis();

        IeltsTurnService.completeTurn(
            CompleteTurnCommand.builder()
                .setTurnId(turnId)
                .setCompletedAtMs(completedAt)
                .setReason(reason)
                .build()
        );

        IeltsTurnAssemblerService.completeTurn(
            turnId,
            completedAt,
            reason
        );

        session.clearActiveTurnId();
        SessionStateStore.save(session);
    }


    private void advanceExamState(
            IeltsMockSceneSession session,
            IeltsExamEvent event) {

        IeltsExamState nextState =
            IeltsExamFlowService.transition(
                session.getExamState(),
                event
            );

        session.setExamState(nextState);
        SessionStateStore.save(session);
    }


    // ========================================================
    // 3. 考试结束后异步触发评分
    // ========================================================

    protected void handleSessionCompleted(
            String localSessionId) {

        IeltsMockSceneSession session =
            SessionStateStore.get(localSessionId);

        String attemptId =
            session.getAttemptId();


        // 结束连续 PCM 流，冻结尚未完成的 post-roll
        IeltsTurnAssemblerService.finishStream(
            attemptId
        );


        // Attempt 从采集中进入 Finalizing
        IeltsAttemptService.finalizeCollection(
            attemptId
        );


        // 先完成会话额度结算
        UsageQuotaService.settle(
            localSessionId
        );


        // 异步评分，不阻塞会话完成请求
        IeltsScoringOrchestrator
            .scoreAsync(attemptId)
            .thenAccept(report -> {

                IeltsAttemptService.saveReport(
                    attemptId,
                    report
                );

                IeltsLearningRecordService
                    .saveAttemptSummary(
                        attemptId,
                        session.getUserId()
                    );

                IeltsLearningRecordService
                    .updateAbilityProfile(
                        session.getUserId(),
                        report
                    );

                SessionWebSocketHandler
                    .sendReportReady(
                        localSessionId,
                        attemptId,
                        report.getScoringStatus()
                    );
            })
            .exceptionally(error -> {

                IeltsAttemptService
                    .markScoringFailed(
                        attemptId,
                        error.getMessage()
                    );

                SessionWebSocketHandler
                    .sendScoringError(
                        localSessionId,
                        attemptId,
                        error.getMessage()
                    );

                return null;
            });
    }
}

```java
@Service
public class IeltsScoringOrchestrator {

    public CompletableFuture<IeltsReport> scoreAsync(
            String attemptId) {

        IeltsAttempt attempt =
            IeltsAttemptService.getAttempt(attemptId);

        IeltsAttemptService.updateScoringStatus(
            attemptId,
            IeltsScoringStatus.SCORING
        );


        // 只选择 Part 1、Part 2、Part 3 的可评分回答
        List<IeltsTurn> scoringTurns =
            attempt.getTurns().stream()
                .filter(IeltsTurn::isScoringEligible)
                .filter(turn ->
                    turn.getPart() >= 1
                    && turn.getPart() <= 3
                )
                .toList();


        // 没有有效回答时不调用模型
        if (scoringTurns.isEmpty()
                || scoringTurns.stream()
                    .noneMatch(turn ->
                        !turn.getRawTranscript().isBlank()
                    )) {

            IeltsReport report =
                IeltsReportService
                    .buildUnscorableReport(
                        attempt,
                        "没有有效 raw transcript"
                    );

            IeltsAttemptService.saveReport(
                attemptId,
                report
            );

            return CompletableFuture
                .completedFuture(report);
        }


        // 计算每个 Turn 的本地客观指标
        List<TurnScoringInput> turnInputs =
            scoringTurns.stream()
                .map(turn ->
                    TurnScoringInput.builder()
                        .setTurn(turn)
                        .setMetrics(
                            IeltsAcousticFeatureService
                                .calculate(turn)
                        )
                        .build()
                )
                .toList();


        // 第一条异步链路：Qwen FC/LR/GRA Language Evidence
        CompletableFuture<LanguageEvidence>
            languageFuture =
                IeltsTextScoringService
                    .languageEvidence(
                        ScoringInput.from(
                            attempt,
                            turnInputs
                        )
                    );


        // 第二条异步链路：每个 Turn 的讯飞发音证据
        List<CompletableFuture<PronunciationEvidence>>
            pronunciationFutures =
                scoringTurns.stream()
                    .map(turn ->
                        PronunciationEvidenceProvider
                            .evaluate(turn)
                            .orTimeout(
                                60,
                                TimeUnit.SECONDS
                            )
                            .exceptionally(error ->
                                PronunciationEvidence.failure(
                                    error.getMessage()
                                )
                            )
                    )
                    .toList();

        CompletableFuture<Void>
            allPronunciationFuture =
                CompletableFuture.allOf(
                    pronunciationFutures.toArray(
                        CompletableFuture[]::new
                    )
                );


        // 等待语言证据和发音证据后，再调用综合 Judge
        return languageFuture
            .thenCombine(
                allPronunciationFuture,
                (languageEvidence, ignored) -> {

                    List<PronunciationEvidence>
                        pronunciationEvidence =
                            pronunciationFutures.stream()
                                .map(
                                    CompletableFuture::join
                                )
                                .toList();

                    return HolisticJudgeInput.builder()
                        .setAttempt(attempt)
                        .setTurnInputs(turnInputs)
                        .setLanguageEvidence(
                            languageEvidence
                        )
                        .setPronunciationEvidence(
                            pronunciationEvidence
                        )
                        .setDataQualityWarnings(
                            DataQualityPolicy.collect(
                                turnInputs,
                                pronunciationEvidence
                            )
                        )
                        .build();
                }
            )
            .thenCompose(
                IeltsTextScoringService::holisticJudge
            )
            .thenApply(judgeResult -> {

                DimensionResult fc =
                    validateDimension(
                        "FC",
                        judgeResult.getFluencyCoherence()
                    );

                DimensionResult lr =
                    validateDimension(
                        "LR",
                        judgeResult.getLexicalResource()
                    );

                DimensionResult gra =
                    validateDimension(
                        "GRA",
                        judgeResult
                            .getGrammaticalRangeAccuracy()
                    );

                DimensionResult pronunciation =
                    validateDimension(
                        "Pronunciation",
                        judgeResult.getPronunciation()
                    );


                // 最终 Overall 由 Java 固定规则计算
                BandResult bandResult =
                    IeltsBandCalculator.calculate(
                        fc.getBand(),
                        lr.getBand(),
                        gra.getBand(),
                        pronunciation.getBand()
                    );


                IeltsReport report =
                    bandResult.getOverallBand() == null
                        ? IeltsReportService
                            .buildPartialReport(
                                attempt,
                                judgeResult,
                                bandResult
                            )
                        : IeltsReportService
                            .buildCompleteReport(
                                attempt,
                                judgeResult,
                                bandResult
                            );

                IeltsAttemptService.saveReport(
                    attemptId,
                    report
                );

                return report;
            })
            .exceptionally(error -> {

                IeltsReport report =
                    IeltsReportService
                        .buildUnscorableReport(
                            attempt,
                            error.getMessage()
                        );

                IeltsAttemptService.saveReport(
                    attemptId,
                    report
                );

                return report;
            });
    }
}
```
